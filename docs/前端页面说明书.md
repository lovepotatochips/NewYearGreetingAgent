#### 1. 发送消息流程

```
用户输入 → 点击发送
    ↓
添加用户消息到列表
    ↓

    ↓

    ↓

    ↓

    ↓
滚动到底部
    ↓
清除加载状态
```

**代码实现**：
```javascript
const sendMessage = async () => {
  // 1. 获取并清空输入
  const userMessage = inputMessage.value.trim()
  inputMessage.value = ''
  
  // 2. 添加用户消息
  messages.value.push({
    id: Date.now(),
    role: 'user',
    content: userMessage,
    created_at: new Date().toISOString()
  })
  
  // 3. 滚动到底部
  scrollToBottom()
  
  // 4. 设置加载状态
  loading.value = true
  
  try {
    // 5. 调用API
    const response = await conversationApi.chat({
      message: userMessage,
      conversation_id: currentConversationId.value
    })
    
    // 6. 处理响应
    if (response && response.conversation && response.message) {
      currentConversationId.value = response.conversation.id
      messages.value.push({
        id: response.message.id || Date.now(),
        role: response.message.role || 'assistant',
        content: response.message.content || '抱歉，我无法理解您的问题。',
        created_at: response.message.created_at || new Date().toISOString()
      })
    }
    
    scrollToBottom()
  } catch (error) {
    ElMessage.error('发送失败，请重试')
  } finally {
    loading.value = false
  }
}
```

const generateGreeting = async () => {
  if (!form.value.targetGroup || !form.value.style || !form.value.formatType) {
    ElMessage.warning('请完善选项')
    return
  }
  
  loading.value = true
  
  try {
    const response = await greetingApi.generateGreeting({
      target_group: form.value.targetGroup,

const submitTool = async () => {
  if (!toolForm.value.input) {
    ElMessage.warning('请填写必要信息')
    return
  }
  
  loading.value = true
  
  try {
    const apiMethod = toolApi[currentTool.value.api]
    const params = {}
    
    // 根据工具类型设置参数
    if (currentTool.value.inputType === 'select') {
      params.target_group = toolForm.value.input
      params.amount_type = toolForm.value.input
      params.people_count = parseInt(toolForm.value.input)
      params.date = toolForm.value.input
      params.scenario = toolForm.value.input
      params.question = toolForm.value.input
    } else {
      params.question = toolForm.value.input
      params.scenario = toolForm.value.input
      params.people_count = parseInt(toolForm.value.input)
      params.date = toolForm.value.input
    }
    
    // 添加可选参数
    if (currentTool.value.hasSecondInput && toolForm.value.secondInput) {
      params.budget = toolForm.value.secondInput
      params.relationship = toolForm.value.secondInput
      params.taste_preference = toolForm.value.secondInput
      params.region = toolForm.value.secondInput
      params.meaning = toolForm.value.secondInput
    }
    
    if (currentTool.value.hasThirdInput && toolForm.value.thirdInput) {
      params.region = toolForm.value.thirdInput
    }
    
    const response = await apiMethod(params)
    
    // 处理响应
    if (response.answer) {
      result.value = response.answer
    } else if (response.guidance) {
      result.value = response.guidance
    } else if (response.suggestions) {

  } finally {
    loading.value = false
  }
}
